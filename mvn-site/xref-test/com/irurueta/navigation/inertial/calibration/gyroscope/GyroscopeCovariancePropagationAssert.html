16  package com.irurueta.navigation.inertial.calibration.gyroscope;
17  
18  import com.irurueta.algebra.Matrix;
19  
20  import static org.junit.jupiter.api.Assertions.assertEquals;
21  import static org.junit.jupiter.api.Assertions.assertTrue;
22  
23  /**
24   * Checks that the general-case gyroscope calibrators propagate the fitter
25   * covariance to the published parameter covariance correctly.
26   * <p>
27   * The fitter unknowns are b = (bx, by, bz) (only when biases are estimated),
28   * M = Mg + I stored column-wise (m11, m21, m31, m12, m22, m32, m13, m23, m33)
29   * and G stored column-wise (only when G-dependent cross biases are estimated).
30   * The published parameters are:
31   * <pre>
32   * bg = M * b                              (only when biases are estimated)
33   * sx, sy, sz, mxy, mxz, myx, myz, mzx, mzy = entries of M - I
34   * Gg = M * G                              (zero when cross biases are not estimated)
35   * </pre>
36   * This class rebuilds that mapping independently of the calibrators,
37   * differentiates it numerically, and checks that J * Cov * J' equals the
38   * covariance published by the calibrator. A wrong or misplaced entry anywhere
39   * in a calibrator's analytical Jacobian makes the check fail.
40   */
41  final class GyroscopeCovariancePropagationAssert {
42  
43      /**
44       * Number of components of a 3D vector.
45       */
46      private static final int COMPONENTS = 3;
47  
48      /**
49       * Number of entries of a 3x3 matrix.
50       */
51      private static final int MATRIX_ENTRIES = 9;
52  
53      /**
54       * Position of sx, sy, sz, mxy, mxz, myx, myz, mzx, mzy within the column-wise
55       * M block of fitter unknowns.
56       */
57      private static final int[] M_INDICES = {0, 4, 8, 3, 6, 1, 7, 2, 5};
58  
59      /**
60       * Step used for central differences. The mapping is bilinear, so central
61       * differences are exact up to rounding error for any step.
62       */
63      private static final double STEP = 1e-3;
64  
65      /**
66       * Tolerance relative to sqrt(var_i * var_j) for covariance entry (i, j).
67       */
68      private static final double RELATIVE_ERROR = 1e-8;
69  
70      private GyroscopeCovariancePropagationAssert() {
71      }
72  
73      /**
74       * Asserts that the covariance published by a calibrator after a general-case
75       * calibration equals the fitter covariance propagated through a numerically
76       * differentiated Jacobian.
77       *
78       * @param calibrator           calibrator that has finished a general-case calibration.
79       * @param estimatedCovariance  covariance returned by the calibrator.
80       * @param estimatesBias        true when the calibrator estimates biases (bias unknowns
81       *                             and published bg rows are present), false for known-bias
82       *                             calibrators.
83       * @param estimatesCrossBiases true when G-dependent cross biases are estimated.
84       */
85      static void assertCovariancePropagated(
86              final Object calibrator, final Matrix estimatedCovariance, final boolean estimatesBias,
87              final boolean estimatesCrossBiases) {
88          final var fitter = getFitter(calibrator);
89          final var params = getParams(fitter);
90          final var fitterCovariance = getCovariance(fitter);
91  
92          final var inputs = (estimatesBias ? COMPONENTS : 0) + MATRIX_ENTRIES
93                  + (estimatesCrossBiases ? MATRIX_ENTRIES : 0);
94          final var outputs = (estimatesBias ? COMPONENTS : 0) + 2 * MATRIX_ENTRIES;
95          assertEquals(inputs, params.length);
96          assertEquals(inputs, fitterCovariance.getRows());
97          assertEquals(outputs, estimatedCovariance.getRows());
98          assertEquals(outputs, estimatedCovariance.getColumns());
99  
100         final var jacobian = numericalJacobian(params, outputs, estimatesBias, estimatesCrossBiases);
101         final var expected = propagate(jacobian, fitterCovariance);
102 
103         for (var i = 0; i < outputs; i++) {
104             for (var j = 0; j < outputs; j++) {
105                 final var tolerance = RELATIVE_ERROR * Math.sqrt(Math.abs(expected[i][i] * expected[j][j]));
106                 assertEquals(expected[i][j], estimatedCovariance.getElementAt(i, j), tolerance,
107                         "covariance entry (" + i + ", " + j + ")");
108             }
109         }
110 
111         // mzy variance must be non-trivial so that a scaled Jacobian entry
112         // cannot go unnoticed
113         final var mzyIndex = (estimatesBias ? COMPONENTS : 0) + MATRIX_ENTRIES - 1;
114         assertTrue(estimatedCovariance.getElementAt(mzyIndex, mzyIndex) > 0.0);
115     }
116 
117     /**
118      * Evaluates the mapping from fitter unknowns to published parameters.
119      */
120     private static double[] map(final double[] x, final int outputs, final boolean estimatesBias,
121                                 final boolean estimatesCrossBiases) {
122         final var mOffset = estimatesBias ? COMPONENTS : 0;
123         final var gOffset = mOffset + MATRIX_ENTRIES;
124         final var result = new double[outputs];
125         var pos = 0;
126 
127         if (estimatesBias) {
128             // bg = M * b
129             for (var row = 0; row < COMPONENTS; row++) {
130                 var value = 0.0;
131                 for (var k = 0; k < COMPONENTS; k++) {
132                     value += m(x, mOffset, row, k) * x[k];
133                 }
134                 result[pos++] = value;
135             }
136         }
137 
138         // entries of Mg = M - I
139         for (var i = 0; i < MATRIX_ENTRIES; i++) {
140             final var index = M_INDICES[i];
141             final var diagonal = index % (COMPONENTS + 1) == 0;
142             result[pos++] = x[mOffset + index] - (diagonal ? 1.0 : 0.0);
143         }
144 
145         // Gg = M * G stored column-wise, or zero
146         for (var col = 0; col < COMPONENTS; col++) {
147             for (var row = 0; row < COMPONENTS; row++) {
148                 var value = 0.0;
149                 if (estimatesCrossBiases) {
150                     for (var k = 0; k < COMPONENTS; k++) {
151                         value += m(x, mOffset, row, k) * x[gOffset + col * COMPONENTS + k];
152                     }
153                 }
154                 result[pos++] = value;
155             }
156         }
157         return result;
158     }
159 
160     private static double m(final double[] x, final int mOffset, final int row, final int col) {
161         return x[mOffset + col * COMPONENTS + row];
162     }
163 
164     private static double[][] numericalJacobian(final double[] params, final int outputs,
165                                                 final boolean estimatesBias, final boolean estimatesCrossBiases) {
166         final var jacobian = new double[outputs][params.length];
167         for (var j = 0; j < params.length; j++) {
168             final var plus = params.clone();
169             final var minus = params.clone();
170             plus[j] += STEP;
171             minus[j] -= STEP;
172             final var fPlus = map(plus, outputs, estimatesBias, estimatesCrossBiases);
173             final var fMinus = map(minus, outputs, estimatesBias, estimatesCrossBiases);
174             for (var i = 0; i < outputs; i++) {
175                 jacobian[i][j] = (fPlus[i] - fMinus[i]) / (2.0 * STEP);
176             }
177         }
178         return jacobian;
179     }
180 
181     private static double[][] propagate(final double[][] jacobian, final Matrix covariance) {
182         final var outputs = jacobian.length;
183         final var inputs = covariance.getRows();
184         final var jc = new double[outputs][inputs];
185         for (var i = 0; i < outputs; i++) {
186             for (var k = 0; k < inputs; k++) {
187                 var value = 0.0;
188                 for (var l = 0; l < inputs; l++) {
189                     value += jacobian[i][l] * covariance.getElementAt(l, k);
190                 }
191                 jc[i][k] = value;
192             }
193         }
194         final var result = new double[outputs][outputs];
195         for (var i = 0; i < outputs; i++) {
196             for (var j = 0; j < outputs; j++) {
197                 var value = 0.0;
198                 for (var k = 0; k < inputs; k++) {
199                     value += jc[i][k] * jacobian[j][k];
200                 }
201                 result[i][j] = value;
202             }
203         }
204         return result;
205     }
206 
207     private static Object getFitter(final Object calibrator) {
208         try {
209             final var field = calibrator.getClass().getDeclaredField("fitter");
210             field.setAccessible(true);
211             return field.get(calibrator);
212         } catch (final ReflectiveOperationException e) {
213             throw new AssertionError("Could not read fitter", e);
214         }
215     }
216 
217     private static double[] getParams(final Object fitter) {
218         try {
219             return (double[]) fitter.getClass().getMethod("getA").invoke(fitter);
220         } catch (final ReflectiveOperationException e) {
221             throw new AssertionError("Could not read fitter parameters", e);
222         }
223     }
224 
225     private static Matrix getCovariance(final Object fitter) {
226         try {
227             return (Matrix) fitter.getClass().getMethod("getCovar").invoke(fitter);
228         } catch (final ReflectiveOperationException e) {
229             throw new AssertionError("Could not read fitter covariance", e);
230         }
231     }
