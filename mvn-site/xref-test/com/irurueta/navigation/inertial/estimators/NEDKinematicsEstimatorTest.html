1415             final double oldVn, final double oldVe, final double oldVd, final double latitude, final double height,
1416             final double oldLatitude, final double oldHeight) throws WrongSizeException, RankDeficientMatrixException,
1417             DecomposerException {
1418 
1419         if (timeInterval > 0.0) {
1420             final var oldVebn = new Matrix(3, 1);
1421             oldVebn.setElementAtIndex(0, oldVn);
1422             oldVebn.setElementAtIndex(1, oldVe);
1423             oldVebn.setElementAtIndex(2, oldVd);
1424 
1425             final var vEbn = new Matrix(3, 1);
1426             vEbn.setElementAtIndex(0, vn);
1427             vEbn.setElementAtIndex(1, ve);
1428             vEbn.setElementAtIndex(2, vd);
1429 
1430             final var cBn = c.getMatrix();
1431             final var oldCbn = oldC.getMatrix();
1432 
1433             final var omegaIe = Constants.EARTH_ROTATION_RATE;
1434 
1435             // From (2.123), determine the angular rate of the ECEF frame with respect
1436             // to the ECI frame, resolved about NED
1437             Matrix tmp = new Matrix(3, 1);
1438             tmp.setElementAtIndex(0, Math.cos(oldLatitude));
1439             tmp.setElementAtIndex(1, 0.0);
1440             tmp.setElementAtIndex(2, -Math.sin(oldLatitude));
1441 
1442             final var omegaIen = tmp.multiplyByScalarAndReturnNew(omegaIe);
1443 
1444             // From (5.44), determine the angular rate of the NED frame with respect
1445             // to the ECEF frame, resolved about NED
1446             final var oldRadiiOfCurvature = RadiiOfCurvatureEstimator.estimateRadiiOfCurvatureAndReturnNew(oldLatitude);
1447             final var oldRn = oldRadiiOfCurvature.getRn();
1448             final var oldRe = oldRadiiOfCurvature.getRe();
1449 
1450             final var radiiOfCurvature = RadiiOfCurvatureEstimator.estimateRadiiOfCurvatureAndReturnNew(latitude);
1451             final var rn = radiiOfCurvature.getRn();
1452             final var re = radiiOfCurvature.getRe();
1453 
1454             final var oldOmegaEnN = new Matrix(3, 1);
1455             oldOmegaEnN.setElementAtIndex(0, oldVebn.getElementAtIndex(1) / (oldRe + oldHeight));
1456             oldOmegaEnN.setElementAtIndex(1, -oldVebn.getElementAtIndex(0) / (oldRn + oldHeight));
1457             oldOmegaEnN.setElementAtIndex(2,
1458                     -oldVebn.getElementAtIndex(1) * Math.tan(oldLatitude) / (oldRe + oldHeight));
1459 
1460             final var omegaEnN = new Matrix(3, 1);
1461             omegaEnN.setElementAtIndex(0, vEbn.getElementAtIndex(1) / (re + height));
1462             omegaEnN.setElementAtIndex(1, -vEbn.getElementAtIndex(0) / (rn + height));
1463             omegaEnN.setElementAtIndex(2, -vEbn.getElementAtIndex(1) * Math.tan(latitude) / (re + height));
1464 
1465             // Get coordinate transformation matrix from the old attitude with respect
1466             // to an inertial frame to the new using (5.77)
1467 
1468             //cOldNew = cBn' * (I - timeInterval*skew(omegaIen + 0.5*omegaEnN + 0.5*oldOmegaEnN)) * oldCbn
1469             final var cOldNew = cBn.transposeAndReturnNew().multiplyAndReturnNew(
1470                     Matrix.identity(3, 3).subtractAndReturnNew(
1471                             Utils.skewMatrix(omegaIen.addAndReturnNew(
1472                                     omegaEnN.multiplyByScalarAndReturnNew(0.5)).addAndReturnNew(
1473                                     oldOmegaEnN.multiplyByScalarAndReturnNew(0.5)))
1474                                     .multiplyByScalarAndReturnNew(timeInterval))
1475                             .multiplyAndReturnNew(oldCbn));
1476 
1477             // Calculate the approximate angular rate with respect to an inertial frame
1478             final var alphaIbb = new Matrix(3, 1);
1479             alphaIbb.setElementAtIndex(0, 0.5 * (
1480                     cOldNew.getElementAt(1, 2) - cOldNew.getElementAt(2, 1)));
1481             alphaIbb.setElementAtIndex(1, 0.5 * (
1482                     cOldNew.getElementAt(2, 0) - cOldNew.getElementAt(0, 2)));
1483             alphaIbb.setElementAtIndex(2, 0.5 * (
1484                     cOldNew.getElementAt(0, 1) - cOldNew.getElementAt(1, 0)));
1485 
1486             // Calculate and apply the scaling factor
1487             final var temp = Math.acos(0.5 * (cOldNew.getElementAt(0, 0) 
1488                     + cOldNew.getElementAt(1, 1) 
1489                     + cOldNew.getElementAt(2, 2) - 1.0));
1490             if (temp > SCALING_THRESHOLD) {
1491                 // scaling is 1 if temp is less than this
1492                 alphaIbb.multiplyByScalar(temp / Math.sin(temp));
1493             }
1494 
1495             // Calculate the angular rate
1496             final var omegaIbb = alphaIbb.multiplyByScalarAndReturnNew(1.0 / timeInterval);
1497 
1498             // Calculate the specific force resolved about ECEF-frame axes
1499             // From (5.54),
1500             final var gravity = NEDGravityEstimator.estimateGravityAndReturnNew(oldLatitude, oldHeight);
1501             final var g = gravity.asMatrix();
1502             final var fIbn = vEbn.subtractAndReturnNew(oldVebn);
1503             fIbn.multiplyByScalar(1.0 / timeInterval);
1504             fIbn.subtract(g);
1505             fIbn.add(Utils.skewMatrix(oldOmegaEnN.addAndReturnNew(omegaIen.multiplyByScalarAndReturnNew(2.0)))
1506                     .multiplyAndReturnNew(oldVebn));
1507 
1508             // Calculate the average body-to-NED coordinate transformation
1509             // matrix over the update interval using (5.84) and (5.86)
1510             final var magAlpha = Utils.normF(alphaIbb);
1511             final var skewAlphaIbb = Utils.skewMatrix(alphaIbb);
1512             final Matrix aveCbn;
1513             if (magAlpha > ALPHA_THRESHOLD) {
1514                 // aveCbn = oldCbn * (I + (1 - cos(magAlpha))/magAlpha^2*AlphaIbb +
1515                 //   (1 - sin(magAlpha)/magAlpha)/magAlpha^2*AlphaIbb*AlphaIbb)
1516                 //   -0.5 * skew(oldOmegaEnn + omegaIen)*oldCbn
1517                 aveCbn = oldCbn.multiplyAndReturnNew(
1518                         Matrix.identity(3, 3).addAndReturnNew(
1519                                 skewAlphaIbb.multiplyByScalarAndReturnNew(
1520                                         (1.0 - Math.cos(magAlpha)) / Math.pow(magAlpha, 2.0))
1521                         ).addAndReturnNew(skewAlphaIbb.multiplyByScalarAndReturnNew(
1522                                 (1.0 - Math.sin(magAlpha) / magAlpha) / Math.pow(magAlpha, 2.0))
1523                                 .multiplyAndReturnNew(skewAlphaIbb))).subtractAndReturnNew(
1524                         Utils.skewMatrix(oldOmegaEnN.addAndReturnNew(omegaIen))
1525                                 .multiplyByScalarAndReturnNew(0.5)
1526                                 .multiplyAndReturnNew(oldCbn));
1527             } else {
1528                 aveCbn = oldCbn.subtractAndReturnNew(Utils.skewMatrix(oldOmegaEnN.addAndReturnNew(omegaIen))
1529                         .multiplyByScalarAndReturnNew(0.5).multiplyAndReturnNew(oldCbn));
1530             }
1531 
1532             // Transform specific force to body-frame resolving axes using (5.81)
1533             final var fIbb = Utils.inverse(aveCbn).multiplyAndReturnNew(fIbn);
1534 
1535             final var fx = fIbb.getElementAtIndex(0);
1536             final var fy = fIbb.getElementAtIndex(1);
1537             final var fz = fIbb.getElementAtIndex(2);
1538 
1539             final var angularRateX = omegaIbb.getElementAtIndex(0);
1540             final var angularRateY = omegaIbb.getElementAtIndex(1);
1541             final var angularRateZ = omegaIbb.getElementAtIndex(2);
1542 
1543             return new BodyKinematics(fx, fy, fz, angularRateX, angularRateY, angularRateZ);
1544 
1545         } else {
1546             // If the time interval is zero, set the angular rate and specific force to zero
1547             return new BodyKinematics();
1548         }
1549     }
