16  package com.irurueta.navigation.inertial.calibration.generators;
17  
18  import com.irurueta.navigation.LockedException;
19  import com.irurueta.navigation.inertial.BodyKinematicsAndMagneticFluxDensity;
20  import com.irurueta.navigation.inertial.BodyMagneticFluxDensity;
21  import com.irurueta.navigation.inertial.calibration.StandardDeviationBodyMagneticFluxDensity;
22  import com.irurueta.navigation.inertial.calibration.intervals.TriadStaticIntervalDetector;
23  import com.irurueta.navigation.inertial.calibration.noise.AccumulatedMagneticFluxDensityTriadNoiseEstimator;
24  import java.util.ArrayDeque;
25  
26  /**
27   * Generates measurements for the calibration of magnetometers by alternating
28   * static and dynamic intervals where device is kept static or moved.
29   * Generated measurements must be used with magnetometer calibrators based
30   * on the knowledge of position on Earth and time instant.
31   * Such calibrators are the following ones:
32   * - {@link com.irurueta.navigation.inertial.calibration.magnetometer.KnownPositionAndInstantMagnetometerCalibrator}
33   * - {@link com.irurueta.navigation.inertial.calibration.magnetometer.KnownHardIronPositionAndInstantMagnetometerCalibrator}
34   * - {@link com.irurueta.navigation.inertial.calibration.magnetometer.RobustKnownPositionAndInstantMagnetometerCalibrator}
35   * and all its implementations.
36   * - {@link com.irurueta.navigation.inertial.calibration.magnetometer.RobustKnownHardIronPositionAndInstantMagnetometerCalibrator}
37   * and all its implementations.
38   * <p>
39   * Generated measurements contain the average magnetic flux density of each static interval, computed
40   * only with the samples that the (centred window) detector classified as static, and its standard deviation
41   * is the one of the average, that is, the mean standard deviation of the three coordinates divided by the
42   * square root of the number of averaged samples. The initialization period is static, so its samples are
43   * part of the first static interval.
44   */
45  public class MagnetometerMeasurementsGenerator extends
46          MeasurementsGenerator<StandardDeviationBodyMagneticFluxDensity,
47                  MagnetometerMeasurementsGenerator, MagnetometerMeasurementsGeneratorListener,
48                  BodyKinematicsAndMagneticFluxDensity> {
49  
50      /**
51       * Accumulated noise estimator for magnetic flux density measurements.
52       */
53      private final AccumulatedMagneticFluxDensityTriadNoiseEstimator accumulatedEstimator =
54              new AccumulatedMagneticFluxDensityTriadNoiseEstimator();
55  
56      /**
57       * Magnetic flux density samples that have been received but not yet classified by the detector,
58       * which classifies the sample at the centre of its window. Each entry contains x, y, z coordinates
59       * expressed in Teslas (T).
60       */
61      private final ArrayDeque<double[]> pendingSamples = new ArrayDeque<>();
62  
63      /**
64       * Indicates that the sample that completed initialization has not been processed yet.
65       */
66      private boolean initializationSamplePending;
67  
68      /**
69       * Constructor.
70       */
71      public MagnetometerMeasurementsGenerator() {
72          super();
73      }
74  
75      /**
76       * Constructor.
77       *
78       * @param listener listener to handle events raised by this generator.
79       */
80      public MagnetometerMeasurementsGenerator(final MagnetometerMeasurementsGeneratorListener listener) {
81          super(listener);
82      }
83  
84      /**
85       * Resets this generator.
86       *
87       * @throws LockedException if generator is busy.
88       */
89      @Override
90      public void reset() throws LockedException {
91          super.reset();
92  
93          accumulatedEstimator.reset();
94          pendingSamples.clear();
95          initializationSamplePending = false;
96      }
97  
98      /**
99       * Post process provided input sample.
100      * Detector classifies the sample at the centre of its window, so samples are buffered until they
101      * are classified, and only samples classified as static contribute to the accumulated average.
102      * The initialization period is static, so its samples are part of the first static interval.
103      *
104      * @param sample an input sample.
105      * @throws LockedException if generator is busy.
106      */
107     @Override
108     protected void postProcess(final BodyKinematicsAndMagneticFluxDensity sample) throws LockedException {
109         final var b = sample.getMagneticFluxDensity();
110         final var status = staticIntervalDetector.getStatus();
111 
112         if (status == TriadStaticIntervalDetector.Status.INITIALIZING) {
113             accumulatedEstimator.addTriad(b.getBx(), b.getBy(), b.getBz());
114             return;
115         }
116 
117         if (status == TriadStaticIntervalDetector.Status.INITIALIZATION_COMPLETED && initializationSamplePending) {
118             // sample that completed initialization belongs to the initialization period
119             initializationSamplePending = false;
120             accumulatedEstimator.addTriad(b.getBx(), b.getBy(), b.getBz());
121             return;
122         }
123 
124         if (status != TriadStaticIntervalDetector.Status.INITIALIZATION_COMPLETED
125                 && status != TriadStaticIntervalDetector.Status.STATIC_INTERVAL
126                 && status != TriadStaticIntervalDetector.Status.DYNAMIC_INTERVAL) {
127             return;
128         }
129 
130         pendingSamples.addLast(new double[]{b.getBx(), b.getBy(), b.getBz()});
131 
132         if (status == TriadStaticIntervalDetector.Status.INITIALIZATION_COMPLETED) {
133             // first classified sample has not been reached yet
134             return;
135         }
136 
137         // oldest pending sample is the one that has just been classified
138         final var classified = pendingSamples.pollFirst();
139 
140         if (status == TriadStaticIntervalDetector.Status.STATIC_INTERVAL) {
141             accumulatedEstimator.addTriad(classified[0], classified[1], classified[2]);
142         }
143     }
144 
145     /**
146      * Gets corresponding acceleration triad from provided input sample.
147      * This method must store the result into {@link #triad}.
148      *
149      * @param sample input sample.
150      */
151     @Override
152     protected void getAccelerationTriadFromInputSample(final BodyKinematicsAndMagneticFluxDensity sample) {
153         sample.getKinematics().getSpecificForceTriad(triad);
154     }
155 
156     /**
157      * Handles a static-to-dynamic interval change.
158      *
159      * @param accumulatedAvgX average x-coordinate of measurements during last
160      *                        static period expressed in meters per squared
161      *                        second (m/s^2).
162      * @param accumulatedAvgY average y-coordinate of specific force during last
163      *                        static period expressed in meters per squared
164      *                        second (m/s^2).
165      * @param accumulatedAvgZ average z-coordinate of specific force during last
166      *                        static period expressed in meters per squared
167      *                        second (m/s^2).
168      * @param accumulatedStdX standard deviation of x-coordinate of measurements
169      *                        during last static period expressed in meters per
170      *                        squared second (m/s^2).
171      * @param accumulatedStdY standard deviation of y-coordinate of measurements
172      *                        during last static period expressed in meters per
173      *                        squared second (m/s^2).
174      * @param accumulatedStdZ standard deviation of z-coordinate of measurements
175      *                        during last static period expressed in meters per
176      *                        squared second (m/s^2).
177      */
178     @Override
179     protected void handleStaticToDynamicChange(
180             final double accumulatedAvgX, final double accumulatedAvgY, final double accumulatedAvgZ,
181             final double accumulatedStdX, final double accumulatedStdY, final double accumulatedStdZ) {
182 
183         // accumulated estimator only contains samples classified as static
184         final var n = accumulatedEstimator.getNumberOfProcessedSamples();
185         if (!isStaticIntervalSkipped() && n > 0) {
186 
187             // the standard deviation of the average of N static samples is sigma / sqrt(N), where
188             // sigma is the mean standard deviation of the three coordinates
189             final var avgStd = (accumulatedEstimator.getStandardDeviationX()
190                     + accumulatedEstimator.getStandardDeviationY()
191                     + accumulatedEstimator.getStandardDeviationZ()) / 3.0;
192 
193             final var measurement = new StandardDeviationBodyMagneticFluxDensity(
194                     new BodyMagneticFluxDensity(accumulatedEstimator.getAvgX(), accumulatedEstimator.getAvgY(),
195                             accumulatedEstimator.getAvgZ()), avgStd / Math.sqrt(n));
196 
197             if (listener != null) {
198                 listener.onGeneratedMeasurement(this, measurement);
199             }
200         }
201 
202         try {
203             accumulatedEstimator.reset();
204         } catch (final LockedException ignore) {
205             // never happens
206         }
207     }
208 
209     /**
210      * Handles a dynamic-to-static interval change.
211      */
212     @Override
213     protected void handleDynamicToStaticChange() {
214         // no action needed.
215     }
216 
217     /**
218      * Handles an initialization completion.
219      */
220     @Override
221     protected void handleInitializationCompleted() {
222         pendingSamples.clear();
223         initializationSamplePending = true;
224     }
225 
226     /**
227      * Handles an error during initialization.
228      */
229     @Override
230     protected void handleInitializationFailed() {
231         pendingSamples.clear();
232         initializationSamplePending = false;
233         try {
234             accumulatedEstimator.reset();
235         } catch (final LockedException ignore) {
236             // no action needed
237         }
238     }
239 }
